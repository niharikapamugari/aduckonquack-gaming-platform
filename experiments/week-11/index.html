🔷 Week 11: TypeScript Classes, Interfaces & Functions

Strongly typed object-oriented architecture modeling games, players, repositories, and calculation logic

TypeScript Source (game_system.ts)

TypeScript 5.x
// Interfaces & Enums export enum GameGenre { ARCADE, PUZZLE, RACING... } export interface IGame { readonly id: string; title: string; genre: GameGenre; calculateScore(base: number, mult: number): number; } export interface IPlayer { username: string; studentId: string; level: number; totalScore: number; addScore(points: number): void; } // Abstract & Concrete Classes export abstract class BaseGame implements IGame { constructor( public readonly id: string, public title: string, public genre: GameGenre ) {} public abstract calculateScore(base: number, mult: number): number; } export class ArcadeGame extends BaseGame { public calculateScore(base: number, mult: number): number { return Math.round(base * mult * 1.5); } } // Generics & Functions export class GameRepository<T extends { id: string }> { private items: Map<string, T> = new Map(); public add(item: T): void { this.items.set(item.id, item); } } export function filterGamesByRating(games: IGame[], min: number): IGame[] { return games.filter(g => g.rating >= min); }
